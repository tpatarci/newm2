```cpp
std::vector<std::string> xdgConfigDirs() {
    const char* dirs = std::getenv("XDG_CONFIG_DIRS");
    std::vector<std::string> result;
    if (dirs && dirs[0] != '\0') {
        std::istringstream ss(dirs);
        std::string dir;
        while (std::getline(ss, dir, ':')) {
            if (!dir.empty() && dir[0] == '/') result.push_back(dir);
        }
    }
    if (result.empty()) result.push_back("/etc/xdg");
    return result;
}
```

From `src/Config.cpp` (lines 84-130) -- `Config::applyFile`, the exact shape `DesktopEntry::parseFile`'s line-reading loop must follow (trim, comment/blank skip, warn-and-continue on malformed lines, never abort):
```cpp
void Config::applyFile(const std::string& path) {
    std::ifstream file(path);
    if (!file.is_open()) return;
    std::string line; int lineNum = 0;
    while (std::getline(file, line)) {
        ++lineNum;
        if (line.size() > 4096) { std::fprintf(stderr, "wm2: warning: config line %d: line too long, skipping\n", lineNum); continue; }
        auto start = line.find_first_not_of(" \t");
        if (start == std::string::npos) continue;
        if (line[start] == '#') continue;
        auto end = line.find_last_not_of(" \t\r\n");
        line = line.substr(start, end - start + 1);
        auto eq = line.find('=');
        if (eq == std::string::npos) { std::fprintf(stderr, "wm2: warning: config line %d: missing '='\n", lineNum); continue; }
        std::string key = line.substr(0, eq), value = line.substr(eq + 1);
        trim(key); trim(value);
        applyKeyValue(key, value);
    }
}
```

| T-7-01 | Tampering / Elevation of Privilege | `DesktopEntry::parseExec` (src/DesktopEntry.cpp) | high | mitigate | Exec= is tokenized into a fixed argv respecting spec quoting; any unrecognized field code invalidates the whole entry (return std::nullopt) rather than passing a raw string to a shell; the resulting argv is designed to be fed directly to `execvp()` in a later plan, with no shell interpreter ever in the launch path for this source |
