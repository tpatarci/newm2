```cpp
#define MenuMask        ( ButtonMask | ButtonMotionMask | ExposureMask )
#define MenuGrabMask    ( ButtonMask | ButtonMotionMask | StructureNotifyMask )

void WindowManager::menu(XButtonEvent *e) {
    if (e->window == m_menuWindow) return;
    std::vector<Client*> clients;
    for (const auto& hc : m_hiddenClients) clients.push_back(hc.get());
    int nh = static_cast<int>(clients.size()) + 1;
    int n = static_cast<int>(clients.size()) + 1;
    int mx = DisplayWidth(display(), m_screenNumber) - 1;
    int my = DisplayHeight(display(), m_screenNumber) - 1;
    bool allowExit = ((e->x > mx - 3) && (e->y > my - 3));
    if (allowExit) n += 1;

    auto menuLabelFn = [&](int idx) -> const char* {
        if (idx == 0) return m_menuCreateLabel;
        if (allowExit && idx > static_cast<int>(clients.size())) return "[Exit wm2]";
        return clients[idx - 1]->label().c_str();
    };
    // width-measurement loop (lines 130-140) uses menuLabelFn(i) for i in [0, n)
    // ... geometry/clamp/warp (lines 142-163) ...
    // XMoveResizeWindow/XMapRaised/XftDraw bind/attemptGrab on m_menuWindow (lines 164-180)
    // event loop (lines 186-297): switch(event.type) { ButtonPress: break; ButtonRelease: {...
    //   sel = y / entryHeight; ... selecting = sel; done = true; } break;
    //   MotionNotify: if (!drawn) break; ... selecting = y / entryHeight; ... (highlight redraw) break;
    //   Expose: clear+border+labels via menuLabelFn(i); labels with i >= nh are right-aligned
    //     (this is the line this plan must change -- currently `if (i >= nh)` right-aligns,
    //     but category rows also have i >= nh and must stay LEFT-aligned; only the true
    //     exit row (i == n-1 when allowExit) should right-align) }
    if (selecting == n - 1 && allowExit) { m_signalled = 1; return; }
    if (selecting >= 0) {
        if (selecting == 0) spawn();
        else if (selecting < nh) clients[selecting - 1]->unhide(true);
        else if (selecting < n) { clients[selecting - 1]->mapRaised(); clients[selecting - 1]->ensureVisible(); }
    }
}
```
